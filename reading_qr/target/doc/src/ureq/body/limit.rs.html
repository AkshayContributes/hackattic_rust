11impl<R> LimitReader<R> {

13        LimitReader {
14            reader,
15            limit,
16            left: limit,
17        }
18    }

21impl<R: io::Read> io::Read for LimitReader<R> {
22    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
23        if self.left == 0 {
24            return Err(Error::BodyExceedsLimit(self.limit).into_io());
25        }
26
27        // The max buffer size is usize, which may be 32 bit.
28        let max = (self.left.min(usize::MAX as u64) as usize).min(buf.len());
29
30        let n = self.reader.read(&mut buf[..max])?;
31
32        self.left -= n as u64;
33
34        Ok(n)
35    }
36}
