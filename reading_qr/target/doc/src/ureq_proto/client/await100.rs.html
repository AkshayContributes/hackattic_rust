11impl Call<Await100> {
12    /// Attempt to read a 100-continue response.
13    ///
14    /// Tries to interpret bytes sent by the server as a 100-continue response. The expect-100 mechanic
15    /// means we hope the server will give us an indication on whether to upload a potentially big
16    /// request body, before we start doing it.
17    ///
18    /// * If the server supports expect-100, it will respond `HTTP/1.1 100 Continue\r\n\r\n`, or
19    ///   some other response code (such as 403) if we are not allowed to post the body.
20    /// * If the server does not support expect-100, it will not respond at all, in which case
21    ///   we will proceed to sending the request body after some timeout.
22    ///
23    /// The results are:
24    ///
25    /// * `Ok(0)` - not enough data yet, continue waiting (or `proceed()` if you think we waited enough)
26    /// * `Ok(n)` - `n` number of input bytes were consumed. Call `proceed()` next
27    /// * `Err(e)` - some error that is not recoverable
28    pub fn try_read_100(&mut self, input: &[u8]) -> Result<usize, Error> {
29        // Try parsing a status line without any headers. The line we are looking for is:
30        //
31        //   HTTP/1.1 100 Continue\r\n\r\n
32        //
33        // There should be no headers.
34        match try_parse_response::<0>(input) {
35            Ok(v) => match v {
36                Some((input_used, response)) => {
37                    let status = response.status();
38
39                    if status == StatusCode::CONTINUE {
40                        // Got 100 Continue - proceed to send the body
41                        self.inner.await_100_continue = false;
42                        assert!(self.inner.state.writer.has_body());
43                        Ok(input_used)
44                    } else if status == StatusCode::SWITCHING_PROTOCOLS {
45                        // 101 Switching Protocols - don't send body, mark connection, proceed to recv response
46                        // https://curl.se/mail/lib-2004-08/0002.html
47                        self.inner.await_100_continue = false;
48                        self.inner.close_reason.push(CloseReason::ProtocolSwitch);
49                        self.inner.state.writer = BodyWriter::new_none();
50                        Ok(0)
51                    } else if status.is_informational() {
52                        // Other 1xx (102 Processing, 103 Early Hints, etc.)
53                        // Consume transparently and continue waiting for 100 or timeout.
54                        // Do NOT set await_100_continue to false - keep waiting.
55                        Ok(input_used)
56                    } else {
57                        // Non-informational response (2xx-5xx) - don't send body
58                        // https://curl.se/mail/lib-2004-08/0002.html
59                        self.inner.await_100_continue = false;
60                        self.inner.close_reason.push(CloseReason::Not100Continue);
61                        self.inner.state.writer = BodyWriter::new_none();
62                        Ok(0)
63                    }
64                }
65                // Not enough input yet.
66                None => Ok(0),
67            },
68            Err(e) => {
69                self.inner.await_100_continue = false;
70
71                if e == Error::HttpParseTooManyHeaders {
72                    // We encountered headers after the status line. That means the server did
73                    // not send 100-continue, and also continued to produce an answer before we
74                    // sent the body. Regardless of what the answer is, we must not send the body.
75                    // A 200-answer would be nonsensical given we haven't yet sent the body.
76                    //
77                    // We do however want to receive the response to be able to provide
78                    // the Response<()> to the user. Hence this is not considered an error.
79                    self.inner.close_reason.push(CloseReason::Not100Continue);
80                    self.inner.state.writer = BodyWriter::new_none();
81                    Ok(0)
82                } else {
83                    Err(e)
84                }
85            }
86        }
87    }
88
89    /// Tell if there is any point in waiting for more data from the server.
90    ///
91    /// Becomes `false` as soon as `try_read_100()` got enough data to determine what to do next.
92    /// This might become `false` even if `try_read_100` returns `Ok(0)`.
93    ///
94    /// If this returns `false`, the user should continue with `proceed()`.
95    pub fn can_keep_await_100(&self) -> bool {
96        self.inner.await_100_continue
97    }
98
99    /// Proceed to the next state.
100    pub fn proceed(self) -> Result<Await100Result, Error> {
101        // We can always proceed out of Await100
102
103        if self.inner.state.writer.has_body() {
104            // TODO(martin): do i need this?
105            // call.inner.call.analyze_request()?;
106            let call = Call::wrap(self.inner);
107            Ok(Await100Result::SendBody(call))
108        } else {
109            Ok(Await100Result::RecvResponse(Call::wrap(self.inner)))
110        }
111    }
112}
