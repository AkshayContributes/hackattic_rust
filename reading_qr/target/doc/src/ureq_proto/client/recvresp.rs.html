13impl Call<RecvResponse> {
14    /// Try reading a response from the input.
15    ///
16    /// * `allow_partial_redirect` - if `true`, we can accept to find the `Location` header
17    ///   and proceed without reading the entire header. This is useful for broken servers that
18    ///   don't send an entire \r\n at the end of the preamble.
19    ///
20    /// The `(usize, Option<Response()>)` is `(input amount consumed, response`).
21    ///
22    /// Notice that it's possible that we get an `input amount consumed` despite not returning
23    /// a `Some(Response)`. This can happen if the server returned a 100-continue, and due to
24    /// timing reasons we did not receive it while we were in the `Await100` call state. This
25    /// "spurios" 100 will be discarded before we parse the actual response.
26    pub fn try_response(
27        &mut self,
28        input: &[u8],
29        allow_partial_redirect: bool,
30    ) -> Result<(usize, Option<Response<()>>), Error> {
31        let maybe_response = self.do_try_response(input, allow_partial_redirect)?;
32
33        let (input_used, response) = match maybe_response {
34            Some(v) => v,
35            // Not enough input for a full response yet
36            None => return Ok((0, None)),
37        };
38
39        // Handle all 1xx informational responses according to RFC 9110.
40        // These can happen in several scenarios:
41        // 1. A "delayed" 100-continue when we used Expect: 100-continue but the server
42        //    did not produce the response in time while we were in the Await100 state.
43        // 2. An unsolicited 1xx response from a server that sends it regardless of
44        //    whether the client requested it (e.g., on a regular GET request).
45        // 3. Other informational responses like 102 Processing or 103 Early Hints.
46        //
47        // According to RFC 9110, all 1xx responses are informational and should be
48        // consumed by clients, with the exception of 101 Switching Protocols.
49        let status = response.status();
50
51        if status == StatusCode::CONTINUE {
52            // Consume 100 Continue and clear the await flag for delayed 100-continue
53            self.inner.await_100_continue = false;
54            return Ok((input_used, None));
55        } else if status == StatusCode::SWITCHING_PROTOCOLS {
56            // 101 is special: mark connection so it won't be returned to pool,
57            // but return the response to the caller (don't consume it).
58            self.inner.close_reason.push(CloseReason::ProtocolSwitch);
59            // Fall through to return the response
60        } else if status.is_informational() {
61            // Other 1xx (102 Processing, 103 Early Hints, etc.): consume and wait for final response
62            return Ok((input_used, None));
63        }
64
65        self.inner.status = Some(response.status());
66        // We want the last Location header.
67        self.inner.location = response
68            .headers()
69            .get_all(header::LOCATION)
70            .into_iter()
71            .next_back()
72            .cloned();
73
74        if response.headers().iter().has(header::CONNECTION, "close") {
75            self.inner
76                .close_reason
77                .push(CloseReason::ServerConnectionClose);
78        }
79
80        Ok((input_used, Some(response)))
81    }
82
83    /// Try reading response headers
84    ///
85    /// A response is only possible once the `input` holds all the HTTP response
86    /// headers. Before that this returns `None`. When the response is succesfully read,
87    /// the return value `(usize, Response<()>)` contains how many bytes were consumed
88    /// of the `input`.
89    fn do_try_response(
90        &mut self,
91        input: &[u8],
92        allow_partial_redirect: bool,
93    ) -> Result<Option<(usize, Response<()>)>, Error> {
94        // ~3k for 100 headers
95        let (input_used, response) = match try_parse_response::<MAX_RESPONSE_HEADERS>(input)? {
96            Some(v) => v,
97            None => {
98                // The caller decides whether to allow a partial parse.
99                if !allow_partial_redirect {
100                    return Ok(None);
101                }
102
103                // TODO(martin): I don't like this code. The mission is to be correct HTTP/1.1
104                // and this is a hack to allow for broken servers.
105                //
106                // As a special case, to handle broken servers that does a redirect without
107                // the final trailing \r\n, we try parsing the response as partial, and
108                // if it is a redirect, we can allow the request to continue.
109                let Some(mut r) = try_parse_partial_response::<MAX_RESPONSE_HEADERS>(input)? else {
110                    return Ok(None);
111                };
112
113                // A redirection must have a location header.
114                let is_complete_redirection =
115                    r.status().is_redirection() && r.headers().contains_key(header::LOCATION);
116
117                if !is_complete_redirection {
118                    return Ok(None);
119                }
120
121                // Insert a synthetic connection: close, since the connection is
122                // not valid after using a partial request.
123                debug!("Partial redirection response, insert fake connection: close");
124                r.headers_mut()
125                    .insert(header::CONNECTION, HeaderValue::from_static("close"));
126
127                (input.len(), r)
128            }
129        };
130
131        log_data(&input[..input_used]);
132
133        let http10 = response.version() == Version::HTTP_10;
134        let status = response.status().as_u16();
135
136        if status == StatusCode::CONTINUE {
137            // There should be no headers for this response.
138            if !response.headers().is_empty() {
139                return Err(Error::HeadersWith100);
140            }
141
142            return Ok(Some((input_used, response)));
143        }
144
145        let force_recv = self.inner.force_recv_body;
146        let recv_body_mode = BodyReader::for_response(
147            http10,
148            self.inner.request.method(),
149            status,
150            force_recv,
151            response.headers(),
152        )?;
153
154        self.inner.state.reader = Some(recv_body_mode);
155
156        Ok(Some((input_used, response)))
157    }

159    /// Tell if we have finished receiving the response.
160    pub fn can_proceed(&self) -> bool {
161        self.inner.state.reader.is_some()
162    }
163
164    /// Tell if response body is closed delimited
165    ///
166    /// HTTP/1.0 does not have `content-length` to serialize many requests over the same
167    /// socket. Instead it uses socket close to determine the body is finished.
168    fn is_close_delimited(&self) -> bool {
169        let rbm = self.inner.state.reader.as_ref().unwrap();
170        matches!(rbm, BodyReader::CloseDelimited)
171    }
172
173    /// Proceed to the next state.
174    ///
175    /// This returns `None` if we have not finished receiving the response. It is guaranteed that if
176    /// `can_proceed()` returns true, this will return `Some`.
177    pub fn proceed(mut self) -> Option<RecvResponseResult> {
178        if !self.can_proceed() {
179            return None;
180        }
181
182        let has_response_body = self.inner.state.need_response_body();
183
184        if has_response_body {
185            if self.is_close_delimited() {
186                self.inner
187                    .close_reason
188                    .push(CloseReason::CloseDelimitedBody);
189            }
190
191            Some(RecvResponseResult::RecvBody(Call::wrap(self.inner)))
192        } else {
193            Some(if self.inner.is_redirect() {
194                RecvResponseResult::Redirect(Call::wrap(self.inner))
195            } else {
196                RecvResponseResult::Cleanup(Call::wrap(self.inner))
197            })
198        }
199    }

201    /// Convert the state to receive a body despite method.
202    ///
203    /// Methods like HEAD and CONNECT should not have attached bodies.
204    /// Some broken APIs use bodies anyway and this is an escape hatch to
205    /// interoperate with such services.
206    pub fn force_recv_body(&mut self) {
207        self.inner.force_recv_body = true;
208    }
