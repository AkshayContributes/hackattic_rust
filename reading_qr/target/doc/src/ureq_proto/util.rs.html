5pub(crate) fn find_crlf(b: &[u8]) -> Option<usize> {
6    let cr = b.iter().position(|c| *c == b'\r')?;
7    let maybe_lf = b.get(cr + 1)?;
8    if *maybe_lf == b'\n' { Some(cr) } else { None }
9}

11pub(crate) fn compare_lowercase_ascii(a: &str, lowercased: &str) -> bool {
12    if a.len() != lowercased.len() {
13        return false;
14    }
15
16    for (a, b) in a.chars().zip(lowercased.chars()) {
17        if !a.is_ascii() {
18            return false;
19        }
20        let norm = a.to_ascii_lowercase();
21        if norm != b {
22            return false;
23        }
24    }
25
26    true
27}

31impl<'a> Writer<'a> {
32    pub(crate) fn new(output: &'a mut [u8]) -> Writer<'a> {
33        Self(Cursor::new(output))
34    }
35
36    pub fn len(&self) -> usize {
37        self.0.position() as usize
38    }
39
40    pub fn available(&self) -> usize {
41        self.0.get_ref().len() - self.len()
42    }
43
44    pub(crate) fn try_write(&mut self, block: impl Fn(&mut Self) -> io::Result<()>) -> bool {
45        let pos = self.0.position();
46        let success = (block)(self).is_ok();
47        if !success {
48            self.0.set_position(pos);
49        }
50        success
51    }
52}

73pub(crate) fn log_data(data: &[u8]) {
74    for row in data.chunks(CHARS_PER_ROW) {

79struct Row<'a>(&'a [u8]);
80
81impl<'a> fmt::Debug for Row<'a> {
82    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
83        for i in 0..CHARS_PER_ROW {
84            if let Some(v) = self.0.get(i) {
85                write!(f, "{}", HEX[*v as usize])?
86            } else {
87                write!(f, "--")?;
88            }
89            if i % 2 == 1 {
90                write!(f, " ")?;
91            }
92        }
93        write!(f, " ")?;
94        for i in 0..CHARS_PER_ROW {
95            if let Some(v) = self.0.get(i) {
96                if v.is_ascii_alphanumeric() || v.is_ascii_punctuation() {
97                    write!(f, "{}", *v as char)?;
98                } else {
99                    write!(f, ".")?;
100                }
101            } else {
102                write!(f, ".")?;
103            }
104        }
105        Ok(())
106    }
107}

128/// Simple impl of an array behaving like a vec.
129#[derive(Clone)]
130pub struct ArrayVec<T, const N: usize> {
131    len: usize,
132    arr: [T; N],
133}
134
135impl<T: PartialEq, const N: usize> PartialEq for ArrayVec<T, N> {
136    fn eq(&self, other: &Self) -> bool {
137        self[..] == other[..]
138    }
139}
140
141impl<T: Eq, const N: usize> Eq for ArrayVec<T, N> {}
142
143impl<T: Default, const N: usize> Default for ArrayVec<T, N> {
144    fn default() -> Self {
145        Self::from_fn(|_| T::default())
146    }
147}
148
149impl<T, const N: usize> Deref for ArrayVec<T, N> {
150    type Target = [T];
151
152    fn deref(&self) -> &Self::Target {
153        &self.arr[..self.len]
154    }
155}
156
157impl<T, const N: usize> DerefMut for ArrayVec<T, N> {
158    fn deref_mut(&mut self) -> &mut Self::Target {
159        &mut self.arr[..self.len]
160    }
161}
162
163impl<T, const N: usize> ArrayVec<T, N> {
164    /// Construct the array.
165    ///
166    /// The function must produces placeholder elements of the type `T`.
167    pub fn from_fn(cb: impl FnMut(usize) -> T) -> Self {
168        Self {
169            len: 0,
170            arr: std::array::from_fn(cb),
171        }
172    }
173
174    /// Add a value T.
175    ///
176    /// Panics if the array is already full.
177    pub fn push(&mut self, value: T) {
178        self.arr[self.len] = value;
179        self.len += 1;
180    }
181
182    /// Try to add a value T to the end of the array.
183    ///
184    /// Returns an error if the array is already full.
185    pub fn try_push(&mut self, value: T) -> Result<(), CapacityError<T>> {
186        if self.len >= N {
187            return Err(CapacityError { element: value });
188        }
189        self.push(value);
190        Ok(())
191    }
192
193    /// Shorten the vec.
194    ///
195    /// This does not drop the elements that are now unused.
196    pub fn truncate(&mut self, len: usize) {
197        assert!(len <= self.len);
198        self.len = len;
199    }
200}

202/// An error returned when capacity of an [`ArrayVec`] is exceeded.
203#[derive(PartialEq, Eq, Clone)]
204pub struct CapacityError<T> {
205    /// The element that triggered this error.
206    element: T,
207}
208
209impl<T> CapacityError<T> {
210    /// Returns the element that triggered this error.
211    pub fn element(self) -> T {
212        self.element
213    }
214}
215
216impl<T> std::error::Error for CapacityError<T> {}
217
218impl<T> fmt::Debug for CapacityError<T> {

220        f.debug_struct("CapacityError").finish()
221    }
222}
